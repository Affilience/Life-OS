function calculateXPRequired(level) {
  // Hybrid formula: Fast early, steady later
  if (level <= 10) {

  } else if (level <= 50) {

### 4. STREAK SYSTEM: "MOMENTUM CHAINS"

#### Concept
Leverage loss aversion and the Zeigarnik effect by creating **Momentum Chains** - visualized as connected stellar nodes that grow stronger with consistency.

#### Why "Momentum" Instead of "Streaks"

**Thematic Fit:**
- Momentum is a physics concept (mass × velocity)
- In space, objects maintain momentum indefinitely
- "Building momentum" = personal growth metaphor
- Less pressure than "don't break the streak"

interface MomentumShield {

// Earning shields
function earnShield(user) {
  if (user.momentumShields.count < user.momentumShields.maxCount) {
    if (user.currentChain % 7 === 0) {  // Every week
      user.momentumShields.count++;
      notify('🛡️ Earned a Momentum Shield! Protect your streak if life happens.');
    }
  }
}

// Using shields
function checkMissedDay(user, activity, today) {
  const yesterday = subDays(today, 1);
  const didYesterday = hasActivity(user, activity, yesterday);

  if (!didYesterday && user[activity].currentChain > 0) {
    // Missed a day!
    if (user.momentumShields.count > 0) {
      // Offer to use shield
      return {
        broken: false,
        shieldUsed: true,
        message: '🛡️ Momentum Shield activated! Your chain continues.'
      };
    } else {
      // No shields, streak breaks
      return {
        broken: true,
        shieldUsed: false,
        message: '💔 Momentum chain broken. But you can start fresh today!'
      };
    }
  }
}
